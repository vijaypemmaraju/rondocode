/* LOCAL ONLY — gitignored, and a production build aliases the loader away.
 *
 * A buildup kit and a drum-and-bass drop, polyrhythms throughout. Original
 * material for pulling apart, not a transcription of anything.
 *
 * The RONDO source is the truth; the rondocode twin is transpiled at load. */
 
import { compile } from '@rondocode/rondo'
 
const rondo = `# BUILDUP -> DROP, 174. Automation, then filth. Polyrhythms throughout.
#
# THE RULE THAT SHAPES ALL OF THIS: a ctrl on a play line is SAMPLED per note,
# so rise 16 steps once per note and cannot ramp inside one. Continuous motion
# comes from an envelope INSIDE the synth driving something that is not
# amplitude:  pitch = swoop -> 90..950  is the whole trick.
#
# The polyrhythms: (p,s) is euclid, {..}%n is polymeter at n steps per bar.
# 5 against 4, 7 against 16, and a 3-step figure at 8 per bar that takes three
# bars to come home.
 
synth whoop
  supersaw pitch detune:.45 mix:.9
  formant
  svf cut res:.6
  * env
  * .3
  pitch = swoop -> 90..950
  swoop = adsr 1.1 .35 .75 .3 ^ 1.6
  cut = knob 1200 300..7000 log
  env = adsr .2 .25 .9 .45
  post
    reverb room:.9 damp:.3 mix:.42
 
synth sweep
  noise
  svf cut res:.9
  * env
  * .15
  cut = climb -> 300..12000
  climb = adsr 24 .1 1 .6 ^ 1.4
  # a REVERSE-cymbal shape: the 20s attack IS the 16-bar build, and r .008
  # slams it shut on the drop instead of fading past it
  env = adsr 20 .5 1 .008
  post
    delay .1875 .3 sync:1 maxtime:1 mix:.22
 
synth swell
  noise
  svf 7000 res:.25
  * env
  * .16
 
  env = adsr 3.4 .1 .9 .3
 
synth tick
  noise
  svf 5200 res:.5 mode:hp
  * env
  * .24
  env = adsr .001 .02 0 .015
 
synth roll
  noise
  svf 3600 res:.35
  * env
  * .3
  tanh
  env = adsr .001 .055 0 .035
 
synth impact
  sine fall
  * amp
  tanh
  * .9
  fall = adsr .002 1.5 0 .35 ^ 2 -> 28..150
  amp = adsr .002 1.7 0 .35
 
# THE CRASH. Nothing cuts a voice at a section boundary — the arrangement only
# decides what STARTS. So the tail is the whole design: decay and release both
# long, and it rings over the drop on its own.
#
# The two plays use DIFFERENT note names for one reason: a retrigger of the
# same note steals its own voice, so a c3 on the drop downbeat would cut the
# c3 still ringing from the build. Noise ignores pitch, so g3 costs nothing
# and buys a second voice.
synth crash
  noise
  svf 7800 res:.12 mode:hp
  * shimmer
  + noise * splash * .5
  tanh
  * .34
  shimmer = adsr .002 2.8 0 2.4
  splash = adsr .001 .06 0 .04
 
# THE REESE. Four detuned saws, a fifth on top for bite, then a phaser, a
# resonant ladder, tube drive, a fold and a bitcrusher. Every stage adds teeth.
synth reese
  saw note
  + saw note * 1.008
  + saw note * .992
  + saw note/2
  + saw note * 1.5 * .35
  phaser rate:.18 depth:.85 feedback:.68 stages:8
  ladder cut res:.72
  shape 3.4 type:tube
  * 1.5
  fold
  bitcrush bits:7 downsample:3
  svf 5400 res:.2
  * env
  tanh
  * .85
  cut = base * wob
  base = knob 900 200..4000 log
  wob = lfo growl tri sync:1 -> .35..1.8
  growl = knob .25 .0625..2
  env = adsr .004 .25 .85 .09
 
synth sub
  sine note
  * env
  tanh
  * .95
  env = adsr .004 .2 .9 .06
 
synth kick
  sine drop
  * amp
  tanh
  * .95
  drop = adsr .001 .08 0 .04 ^ 2 -> 48..190
  amp = adsr .001 .18 0 .06
 
synth snare
  noise
  svf 2200 res:.35
  + sine 190 * body
  * amp
  tanh
  * .55
  body = adsr .001 .07 0 .03
  amp = adsr .001 .13 0 .05
 
synth hat
  noise
  svf 8500 res:.2 mode:hp
  * env
  * .28
  env = adsr .001 .028 0 .02
 
synth rim
  noise
  svf 3000 res:.7
  + sine 420 * body
  * amp
  * .3
  body = adsr .001 .03 0 .02
  amp = adsr .001 .05 0 .03
 
section build 16
 
  play whoop
    c3
    dur: .95
    cut: rise 16 600..7000
    gain: rise 16 .3..0.95
 
  # dur MULTIPLIES the note's whole, it is not a bar count. slow 16 already
  # makes this note 16 bars long, so dur: 16 was holding the gate open for
  # 256 bars and the riser never released at all. dur: 1 ends it exactly on the
  # drop, and sweep's r .008 makes that a cliff.
  play sweep
    c3
    slow 16
    dur: 1
    gain: .45
 
  # dur: 1, not 4 — slow 4 already makes this a four-bar note (the eval warns
  # about this now, and it caught this one)
  play swell
    c3
    slow 4
    dur: 1
    gain: rise 16 .25..0.45
 
  # FIVE against FOUR: five hits at four steps per bar, so the figure walks
  # around the bar and only comes home after five of them
  beat ticks
    {tick ~ ~ tick ~}%4
 
  # 7 hits spread over 16 steps, rotating on odd bars
  beat rims
    <rim(7,16) rim(7,16,3)>
 
  beat rolls
    <~ ~ ~ ~ roll roll roll roll roll*2 roll*2 roll*2 roll*2 roll*4 roll*8 roll*8 roll*16>
 
  beat pulse
    <~ ~ ~ ~ ~ ~ ~ ~ [kick ~ kick ~] [kick ~ kick ~] [kick ~ kick ~] [kick ~ kick ~] [kick ~ kick ~] [kick ~ kick ~] ~ ~>
 
  # QUIET, and short. At gain .95 with dur 2 this was a two-bar sub boom that
  # filled the low end BEFORE the drop and then rang on through it, so the
  # drop arrived into a bass register that was already full: measured, it cost
  # 5 dB of the drop's sub contrast, more than every other factor combined.
  play impact
    <~ ~ ~ ~ ~ ~ ~ ~ ~ ~ ~ ~ ~ ~ ~ c1>
    dur: .75
    gain: .35
 
  # on the downbeat of the LAST bar, so it is already decaying when the drop
  # lands and the drop's own crash stacks on top of it
  play crash
    <~ ~ ~ ~ ~ ~ ~ ~ ~ ~ ~ ~ ~ ~ ~ c3>
    dur: 1
    gain: .9
 
section drop 16
 
  # g3, not c3 — see the synth. Every 4 bars, marking the phrase.
  play crash
    <g3 ~ ~ ~>
    dur: 1
    gain: .8
 
  # the break changes every bar: four variants, one of them a hole
  beat brk
    <[kick ~ ~ ~ snare ~ ~ ~ ~ ~ kick ~ snare ~ ~ ~] [kick ~ ~ kick snare ~ ~ ~ ~ kick ~ ~ snare ~ ~ kick] [kick ~ ~ ~ snare ~ ~ kick ~ ~ kick ~ snare ~ kick ~] [~ ~ ~ ~ snare ~ ~ ~ ~ ~ kick ~ snare ~ snare snare]>
 
  # SEVEN against EIGHT: seven hats at eight steps per bar, home after seven
  beat hats
    {hat ~ hat hat ~ hat ~}%8
 
  # and a third pulse: three ticks at four per bar, home after three
  beat ticks2
    {tick ~ ~}%4
 
  # and a rim at 7 over 16, rotating, so nothing lines up twice
  beat rims2
    <rim(7,16) rim(7,16,2) rim(7,16,4) rim(7,16,6)>
 
  # the reese: the riff moves, and the filter jumps per bar
  play reese
    <[0 ~ ~ 0 ~ 3 ~ ~] [0 ~ 3 ~ ~ 0 ~ ~] [0 ~ ~ 5 ~ 3 ~ 0] [~ ~ 0 ~ 3 ~ 5 ~]>
    scale: e-min
    sub 12
    dur: .42
    base: <600 1600 900 2600>
    gain: .9
 
  play sub
    <[0 ~ ~ 0 ~ 3 ~ ~] [0 ~ 3 ~ ~ 0 ~ ~] [0 ~ ~ 5 ~ 3 ~ 0] [~ ~ 0 ~ 3 ~ 5 ~]>
    scale: e-min
    sub 24
    dur: .4
    gain: 1
 
song build drop
 
# sub:.45, not 1 — a full duck on a break this busy holds the sub down most
# of the bar, which is exactly where the drop's weight is supposed to be
sidechain kick depth:.6 release:.11 reese:.9 sub:.45 hat:.4
 
# makeup is in dB, not a factor. -4.5 dB does by hand what the 0.89 ceiling
# was doing anyway, but UNIFORMLY and before the ceiling, so every gain above
# this line stays live instead of being swallowed
master threshold:-7 ratio:2 attack:20 release:140 makeup:-3.4
 
bpm 174
`
 
const compiled = compile(rondo)
if (!compiled.ok) throw new Error('buildup.ts failed to compile: ' + JSON.stringify(compiled.errors))
 
export default { name: 'buildup + drop', code: compiled.code, rondo }
