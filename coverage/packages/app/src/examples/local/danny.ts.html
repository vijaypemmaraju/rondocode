/* LOCAL example - gitignored. "Oh Danny Boy" (Londonderry Air: traditional
 * melody + Weatherly's 1913 lyrics, both public domain). The melody comes
 * FAITHFULLY from a monophonic voice MIDI (8notes), imported at qErr 0.00 in
 * F major at 100 BPM: every @weight below is the import's own rhythm, and
 * the phrase spans NINE bars via `cycles: 9`. */
const rondo = `# OH DANNY BOY. the melody is a
# faithful MIDI import: the @-weights
# are its real rhythm, and cycles: 9
# lets one phrase run nine bars.
# the vocal bakes on first play.
 
synth keys
  tri note
  * .6
  + h2
  svf 2200 res:.12
  * env
  * .22
  h2s = sine note*2
  h2 = h2s * .18
  env = adsr .02 .4 .7 .5
  post
    chorus rate:.3 depth:.004 mix:.2
    reverb room:.88 damp:.4 mix:.32
 
sing danny voice:barbara
  oh dan-ny boy the pipes the pipes are call-ing _ from glen to glen and down the moun-tain side the sum-mers gone and all the ros-es fall-ing _ tis you tis you must go and i must bide
  <[~@10 e4@2 f4@2 g4@2] [a4@6 g4@2 a4@2 d5@2 c5@2 a4@2] [g4@2 f4@2 d4@6 f4@2 a4@2 a#4@2] [c5@6 d5@2 c5@2 a4@2 f4@2 a4@2] [g4@10 e4@2 f4@2 g4@2] [a4@6 g4@2 a4@2 d5@2 c5@2 a4@2] [g4@2 f4@2 d4@6 e4@2 f4@2 g4@2] [a4@6 a#4@2 a4@2 g4@2 f4@2 e4@2] f4>
  cycles: 9
  gain: .95
  post
    reverb room:.9 damp:.35 mix:.22
 
play keys
  <F F Bb C F F Bb C F>
  dur: .94
  gain: .55
 
cps .4167
`
 
const code = `const keys = synth(({ note, gate, adsr, sine, svf, tri }) => {
  const h2s = sine(note.freq.mul(2))
  const h2 = h2s.mul(0.18)
  const env = adsr(gate, { a: 0.02, d: 0.4, s: 0.7, r: 0.5 })
  return svf(tri(note.freq).mul(0.6).add(h2), 2200, { res: 0.12 }).mul(env).mul(0.22)
}, ({ input, chorus, reverb }) => {
  return ((x) => x.mix(reverb(x, { roomSize: 0.88, damp: 0.4 }), 0.32))(chorus(input, { rate: 0.3, depth: 0.004, mix: 0.2 }))
})
 
p('danny', sing('barbara', 'oh dan-ny boy the pipes the pipes are call-ing _ from glen to glen and down the moun-tain side the sum-mers gone and all the ros-es fall-ing _ tis you tis you must go and i must bide', '<[~@10 e4@2 f4@2 g4@2] [a4@6 g4@2 a4@2 d5@2 c5@2 a4@2] [g4@2 f4@2 d4@6 f4@2 a4@2 a#4@2] [c5@6 d5@2 c5@2 a4@2 f4@2 a4@2] [g4@10 e4@2 f4@2 g4@2] [a4@6 g4@2 a4@2 d5@2 c5@2 a4@2] [g4@2 f4@2 d4@6 e4@2 f4@2 g4@2] [a4@6 a#4@2 a4@2 g4@2 f4@2 e4@2] f4>', { name: 'danny', cycles: 9, post: ({ input, reverb }) => {
  return ((x) => x.mix(reverb(x, { roomSize: 0.9, damp: 0.35 }), 0.22))(input)
} }).gain(0.95))
 
p('keys', chord('<F F Bb C F F Bb C F>').sound('keys').dur(0.94).gain(0.55))
 
setCps(0.4167)
`
 
const example: Example = { name: 'danny boy', rondo, code }
export default example
