import type { Example } from '../index'
 
/* LOCAL example — gitignored, never shipped. Keep iterating here; the app
 * loads ./local/*.ts automatically. Render with:
 *   pnpm tsx packages/server/scripts/render-local.ts \
 *     packages/app/src/examples/local/language.ts 16 out.wav */
const code = `// Porter Robinson — "Language" (main melody), imported from MIDI, rebuilt as a
// BUILD -> DROP. Everything is SCALE-RELATIVE: change 'a major' to move key.
// The reverb BLOOMS into the drop (a driven post param), eq/exciter/ott shape
// the mix, and a four-on-floor + sidechain pump make the drop breathe. 128 BPM.
 
// LEAD — Porter vibrato-saw: 3 detuned saws, ~5.5 Hz pitch vibrato (depth driven
// up on the build), filter opens on the build. Post: eq clears mud + adds an air
// shelf, exciter adds sheen, ott glues, and a DRIVEABLE 'wet' reverb blooms
// dry->wet into the drop (post params are controllable via .ctrl now).
const lead = synth(({ note, gate, param, adsr, saw, noise, sine, svf, shape }) => {
  const cutoff = param('cutoff', 9000, { min: 500, max: 16000, curve: 'log' })
  const vib = param('vib', 0, { min: 0, max: 1 })
  const env = adsr(gate, { a: 0.008, d: 0.3, s: 0.72, r: 0 })
  const f = note.freq.mul(sine(5.5).mul(vib).mul(0.006).add(1))
  const osc = saw(f.mul(0.5)).mul(0.7).mix(saw(f.mul(1.004)), 0.7).mix(saw(f.mul(2.006)), 0.4)
  const filt = svf(osc.add(noise('white').mul(0.012)), cutoff.mul(env.range(0.45, 1)), { res: 0.22 }).mul(env)
  return shape(filt, 1.7, { type: 'tube' }).mul(0.22)
}, ({ input, param, chorus, delay, reverb, eq, exciter, ott, compress }) => {
  const wet = param('wet', 0.04, { min: 0, max: 0.55 }) // .ctrl('wet', …) -> reverb bloom
  const w = input.mix(chorus(input, { rate: 0.5, depth: 0.003, mix: 1 }), 0.15)
  const echoed = w.add(delay(w, 0.351, 0.14).mul(0.12))
  const toned = exciter(eq(echoed, [{ type: 'hp', freq: 170 }, { type: 'highshelf', freq: 7000, gain: 4 }]), { freq: 5000, amount: 0.35 })
  const verb = toned.mix(reverb(toned, { roomSize: 0.92, damp: 0.28 }), wet)
  return compress(ott(verb, { depth: 0.22 }), { threshold: -18, ratio: 2.8, attack: 8, release: 130, makeup: 2 })
}, { unison: 9, detune: 22, spread: 0.9, voices: 20 })
 
// PAD — fat supersaw chord bed. eq de-muds + adds air, ott glues it dense.
const pad = synth(({ note, gate, adsr, saw, svf, lfo }) => {
  const env = adsr(gate, { a: 0.12, d: 0.4, s: 0.85, r: 0.6 })
  const f = note.freq
  const osc = saw(f.mul(0.5)).mul(0.4).mix(saw(f), 0.7).mix(saw(f.mul(1.004)), 0.6).mix(saw(f.mul(2.002)), 0.22)
  return svf(osc, lfo(0.04).range(900, 2800), { res: 0.12 }).mul(env).mul(0.2)
}, ({ input, chorus, shape, eq, ott, reverb }) => {
  const toned = eq(shape(input, 1.4, { type: 'tube' }), [{ type: 'hp', freq: 200 }, { type: 'peak', freq: 450, gain: -3, q: 1.2 }, { type: 'highshelf', freq: 8000, gain: 3 }])
  const w = toned.mix(chorus(toned, { rate: 0.3, depth: 0.005, mix: 1 }), 0.18)
  return ott(w, { depth: 0.3, low: 200, high: 3000 }).mix(reverb(toned, { roomSize: 0.9, damp: 0.4 }), 0.12)
}, { unison: 9, detune: 20, spread: 0.9, voices: 26 })
 
// SUB — simple clean sub bass: sine + a touch of saw, soft-saturated.
const sub = synth(({ note, gate, adsr, sine, saw }) => {
  const env = adsr(gate, { a: 0.014, d: 0.1, s: 0.95, r: 0 })
  return sine(note.freq).mul(0.9).add(saw(note.freq).mul(0.14)).mul(env).mul(0.5).tanh().mul(0.6)
}, ({ input }) => input, { voices: 4 })
 
// SPARKLE — glassy bell an octave over the melody (top.add(7) = +1 octave in a
// 7-note scale), for the drop.
const spark = synth(({ note, gate, adsr, tri, sine, svf }) => {
  const env = adsr(gate, { a: 0.003, d: 0.2, s: 0.12, r: 0.14 })
  const f = note.freq
  const osc = tri(f).mul(0.45).add(sine(f.mul(2.01)).mul(0.3)).add(sine(f.mul(3.01)).mul(0.22)).add(sine(f.mul(5.01)).mul(0.14))
  return svf(osc, 16000, { res: 0.2 }).mul(env).mul(0.4)
}, ({ input, delay, reverb }) => input.add(delay(input, 0.234, 0.26).mul(0.2)).mix(reverb(input, { roomSize: 0.9, damp: 0.24 }), 0.12))
 
// KICK — punchy four-on-floor for the drop. The final .mul() is the real OUTPUT
// gain (.gain() on the pattern is only note velocity, clamped 0..1) — push it
// hard and let the master compressor/limiter tame the peaks.
const kick = synth(({ gate, adsr, sine, shape }) => {
  const amp = adsr(gate, { a: 0.001, d: 0.3, s: 0, r: 0.05 })
  return shape(sine(adsr(gate, { a: 0.001, d: 0.06, s: 0, r: 0.03 }).range(48, 150)).mul(amp), 1.8, { type: 'tube' }).mul(1.8)
})
 
// RISER — white-noise sweep that lifts through the build.
const riser = synth(({ gate, adsr, noise, svf }) => {
  const env = adsr(gate, { a: 13, d: 0, s: 1, r: 0 })
  return svf(noise('white'), env.range(400, 9000), { mode: 'hp', res: 0.4 }).mul(env.pow(2)).mul(0.12)
})
 
// ---- imported melody as SCALE DEGREES in A major (degree 0 = A) ----
const midV = n('<[-3@2 -7@2 -7@2 -7@2 -7@2 -7@2 -7@2 -7@2] [~@2 -7@2 -7@2 -7@2 -7@2 -7@2 -7@2 -7@2] [-5@2 -5@2 -5@2 -6@2 -6@2 -6@2 -6@2 -6@2] [~@2 -8@2 -8@2 -7@2 -7@2 -7@2 -7@2 -7@2] [-3@2 -7@2 -7@2 -7@2 -7@2 -7@2 -7@2 -7@2] [~@2 -7@2 -7@2 -7@2 -7@2 -7@2 -7@2 -7@2] [-5@2 -5@2 -5@2 -6@2 -6@2 -6@2 -6@2 -6@2] [~@2 -8@2 -8@2 -7@2 -7@2 -7@2 -7@2 -7@2]>').scale('a major')
const top = n('<[-3@2 -2@2 -3@2 -2@2 -3@2 -2@2 -3@2 -2@2] [-3@2 0@2 -3@2 0@2 -3@2 0@2 -3@2 0@2] [-3@2 0@2 -3@2 -1@2 -3@2 -1@2 -3@2 -1@2] [-3@2 2@2 -3@2 -2@2 -3@2 -2@2 -3@2 -2@2] [-3@2 -2@2 -3@2 -2@2 -3@2 -2@2 -3@2 -2@2] [-3@2 0@2 -3@2 0@2 -3@2 0@2 -3@2 0@2] [-3@2 0@2 -3@2 -1@2 -3@2 -1@2 -3@2 -1@2] [-3@2 2@2 -3@2 -2@2 -3@2 -2@2 -3@2 -2@2]>').scale('a major')
const upper = stack(midV, top)
// moving bars: dotted-quarter (3 eighths) into an eighth-tied-to-a-half (5
// eighths) — the 3+5 anticipation. @-weights are eighth counts (3+5 = a bar).
const bass = n('<-14 -14 [-12@3 -13@5] [-15@3 -14@5] -14 -14 [-12@3 -13@5] [-15@3 -14@5]>').scale('a major').add(10) // scale-aware transpose
 
// ---- sections ----
// BUILD: melody + pad + sub, everything SWELLING — the filter opens, the vibrato
// deepens, the reverb BLOOMS (driven 'wet'), levels rise, a riser sweeps up. No drums.
const swell = saw.slow(8)
const build = stack(
  upper.sound('lead').ctrl('cutoff', swell.range(2500, 11000)).ctrl('vib', swell.range(0, 1)).ctrl('wet', swell.range(0.05, 0.4)).gain(swell.range(0.4, 0.95)),
  upper.sound('pad').dur(0.95).gain(swell.range(0.5, 1.05)),
  bass.sound('sub').gain(swell.range(0.6, 1)),
  note('c4').slow(8).sound('riser').gain(swell.range(0.2, 1)),
)
// DROP: full energy — bright, wide, wet; a four-on-floor kick + sparkle on top,
// and the sidechain pump (below) makes the whole thing breathe.
const drop = stack(
  upper.sound('lead').ctrl('cutoff', 13000).ctrl('vib', 1).ctrl('wet', 0.42).gain(1),
  upper.sound('pad').dur(0.95).gain(1.1),
  bass.sound('sub').gain(1),
  top.add(7).sound('spark').gain(0.7),
  note('c2*4').sound('kick'), // level lives in the synth's output .mul (gain = velocity only)
)
 
p('language', arrange([8, build], [8, drop], [8, build], [8, drop]))
// pump: the kick (drop only) ducks the melodic + low elements — deeper duck
sidechain('kick', { depth: 0.7, release: 0.09, duck: { lead: 0.5, pad: 0.65, sub: 0.8, spark: 0.35 } })
masterCompress({ threshold: -14, ratio: 2.6, attack: 12, release: 150, makeup: 2 })
setCps(0.5333)
`
 
const example: Example = { name: 'language saw', code }
export default example
