 *
 * 2026-08-07: replaced wholesale with the current working version, then cleaned
 * up in three passes. 26,745 -> 20,831 chars of rondo, 22% smaller, with every
 * section verified event-by-event (a hash of every note's time, pitch,
 * velocity and synth) rather than by eye.
 *
 * 1. THE FIGURES, named once. Eight were repeated 2-6 times each, three of
 *    them 333 characters wide, so changing the riff meant six edits that had
 *    to stay character-identical by hand.
 *
 * 2. THE ARRANGEMENT, said once.
 *      - `intro` and `intro2` shared FOUR whole blocks (three arps + the sub).
 *        Those are `intro_base`; each section carries only its own piano.
 *      - `build2` repeated every one of `build`'s blocks verbatim. It is
 *        `with build` plus the two piano parts that make it build2.
 *      - dropped two commented-out `play` blocks in `intro` that `intro2`
 *        runs for real.
 *      - REMOVED a duplicated `play lead` block in `build` (identical but for
 *        a commented-out gain line). It was doubling the lead rather than
 *        adding a part: build 640 -> 392 onsets, build2 704 -> 456.
 *
 * 3. THE NOTATION ITSELF, 1,707 -> 686 chars across the figures. Every rewrite
 *    was PROVEN by querying both patterns and comparing the event streams,
 *    because two of them are not the identities they look like:
 *      - `[-7 -7 -7 -7]` is FOUR onsets; `-7` is one held note. Runs collapse
 *        to `-7*4`, never to `-7`.
 *      - `[~ -7*7]` is TWO cells, not eight — a `*` subdivides one cell. Mixed
 *        runs need `!` (`[~ -7!7]`), which keeps the cell count.
 *      - `@2` on EVERY cell of a group is doing nothing, since equal weights

 *

# ---------------------------------------------------------------- THE RIG
# The visual follows the ARRANGEMENT, not just the spectrum, so it lands with
# the track instead of merely reacting to it.
#
# \`cycle\` is the bar count straight off the scheduler, so wrapping it over the
# 16-bar song gives the bar we are on and it stays right across a stop and a
# restart. (This example used to derive it as \`time * cps\`, off the renderer's
# own clock, which drifted from the transport the moment you stopped.)
#
#   build   0 -> 1 across bars 0-8. One cold shaft, haze thickening, the
#           batten creeping up. Deliberately restrained: a build that is
#           already a light show leaves the drop nowhere to go.
#   drop    0 -> 1 right at bar 8, when the kick lands. Six beams, strobe,
#           blinders, floor wash, warm hue.
#
# The pump is read from \`duck\` -- the sidechain envelope as APPLIED, rather
# than guessed at from bass energy, which is a different shape.
 
visual
  fn hue(h: f32) -> vec3f {
    let k = abs(fract(h + vec3f(0.0, 0.66667, 0.33333)) * 6.0 - 3.0) - 1.0;
    return clamp(k, vec3f(0.0), vec3f(1.0));
  }
 
  // One volumetric beam. The cone WIDENS with distance and dims with it,
  // which is the whole reason a beam in haze reads as a beam: a constant
  // width line looks like a drawn stroke, not a light.
  //
  // TWO GAUSSIANS, not one. A single wide gaussian is a soft wash -- which is
  // exactly what this looked like. A real shaft through haze is a NARROW hot
  // core sitting inside a much wider, much dimmer halo, and the ratio between
  // them is what reads as intensity. Narrowing a single gaussian instead just
  // makes a dim thin line, because a thinner beam carries less light.
  fn beam(p: vec2f, origin: vec2f, ang: f32, spread: f32) -> f32 {
    let d = vec2f(cos(ang), sin(ang));
    let rel = p - origin;
    let t = dot(rel, d);
    if (t < 0.0) { return 0.0; }
    let perp = length(rel - d * t);
    let w = 0.008 + t * spread;
    let q = (perp * perp) / (w * w);
    let core = exp(-q);
    let halo = exp(-q * 0.055) * 0.30;
    // reaches further than before: the old exp(-t * 0.5) died before the beam
    // crossed the frame, which is half of why they read as blobs near the truss
    return (core + halo) * exp(-t * 0.34);
  }
 
  /* The fixture itself: a hard flare where the beams leave the truss. Without
   * it the shafts appear out of nothing, which is the main tell that this is a
   * gradient rather than a light. */
  fn flare(p: vec2f, at: vec2f, amt: f32) -> f32 {
    let r = length((p - at) * vec2f(1.0, 1.35));
    return (exp(-r * 26.0) + exp(-r * 7.0) * 0.35) * amt;
  }
 
  fn render(uv: vec2f) -> vec4f {
    let asp = res.x / max(res.y, 1.0);
    let p = (uv - 0.5) * vec2f(2.0 * asp, 2.0);
    var col = vec3f(0.0);
 
    // WHERE WE ARE. 16-bar song, 8 build + 8 main, straight off the scheduler.
    let bars = fract(cycle / 16.0) * 16.0;
    let build = clamp(bars / 8.0, 0.0, 1.0);
    let drop = smoothstep(7.90, 8.20, bars);
    // the last bars of the build tighten before it opens
    let tense = smoothstep(5.5, 8.0, bars) * (1.0 - drop);
    // stopped: hold the rig at a dim idle instead of freezing mid-flash
    let live = 0.25 + 0.75 * playing;
 
    let truss = vec2f(0.0, 1.3);
 
    // WHAT EACH BANK FOLLOWS.
    //
    // This used to be hit_ alone, and hit_ is an ONSET SPIKE: it jumps to the
    // note's velocity and falls over ~0.12 s no matter how long the note is.
    // Six fixtures each doing that, four times a bar, is a rig that only ever
    // flashes -- which is what read as jerky.
    //
    // lvl_ is the synth's actual loudness and stays up while a note is held,
    // so it carries the BODY and hit_ only supplies the accent on top. A lead
    // that sustains now glows and swells instead of blinking.
    let leadE = lvl_lead * 2.2 + hit_lead * 0.35;
    let subE  = lvl_sub  * 2.6 + hit_sub  * 0.30;
    let kickE = lvl_kick * 1.6 + hit_kick * 0.55;
 
    // COLOUR follows the spectrum, not a timer. centroid is brightness as one
    // number, so the rig warms as the filter opens through the build and cools
    // when the patch darkens -- the hue is doing something the music does.
    //
    // CALIBRATED, not guessed. Measured on this patch, centroid runs 0..0.45
    // with a mean of 0.038: the energy centre sits near 450 Hz, which is right
    // for something this bass-heavy. A linear map left the rig cold every
    // frame. sqrt spreads the bottom of that range, where all the movement
    // actually is, so the hue travels the way the filter does.
    let warm = clamp(sqrt(max(centroid, 0.0)) * 1.5, 0.0, 1.0);
    let baseHue = 0.60 - warm * 0.52;
 
    // duck is 1 when open and dips on every kick, so 1-duck IS the pump, and
    // it is the envelope the audio is ACTUALLY being multiplied by
    let pump = (1.0 - duck) * drop;
 
    // HAZE. Present from the first bar so the room is never black, thickening
    // through the build and breathing on the real pump once the kick lands.
    let low = smoothstep(1.0, -1.0, p.y);
    // DARKER than before on the drop: the beams got tighter, so the thing
    // they need most is somewhere dark to be bright against.
    let hazeCol = mix(vec3f(0.028, 0.042, 0.095), vec3f(0.050, 0.034, 0.092), drop);
    col += hazeCol * (0.30 + low) * (0.55 + build * 0.7) * (1.0 + bass * 0.55 + pump * 1.3);
 
    // BUILD: one cold shaft, swinging wider and brightening as the section
    // goes. It is the only fixture until the drop, which is the point. Driven
    // by the lead's LEVEL, so it rises with the filter sweep rather than
    // stuttering on every note.
    let sweep = sin(time * 0.22) * (0.10 + build * 0.30);
    col += mix(vec3f(0.35, 0.62, 1.0), vec3f(0.75, 0.85, 1.0), tense)
         * beam(p, truss, -1.5708 + sweep, 0.10)
         * (0.10 + build * 0.45 + leadE * (0.35 + build * 0.7))
         * (1.0 - drop * 0.55) * live;
 
    // DROP: the full bank, in PAIRS -- 0/3 lead, 1/4 sub, 2/5 kick -- so the
    // rig reads as three banks answering three parts rather than six
    // independent lights all doing their own thing.
    //
    // crest is peak-over-rms, so a punchy passage FOCUSES the beams and a
    // dense one lets them bloom. width fans them: a wide stereo image throws
    // the outer pair further out.
    // same calibration story: crest measured 0..0.52, mean 0.006, so a linear
    // read of it never moved the beams at all
    let punch = clamp(sqrt(max(crest, 0.0)) * 1.3, 0.0, 1.0);
    // TIGHT cones. The old 0.095 base spread made every shaft a wash by the
    // time it crossed the frame; a third of that plus the two-gaussian profile
    // is what turns them into shafts. Punch narrows them further.
    let focus = 0.030 - punch * 0.013;
    let fanScale = 0.175 + width * 0.09;
    for (var i: i32 = 0; i < 10; i++) {
      let fi = f32(i);
      let m = i % 3;
      var env = leadE;
      if (m == 1) { env = subE; }
      if (m == 2) { env = kickE; }
      // the sub pair sweeps at a third the rate, so there is slow movement
      // under the fast movement instead of one busy speed
      let rate = select(0.62, 0.21, m == 1);
      let fan = (fi - 4.5) * fanScale;
      // the lead pair TILTS with the melody: note_lead is a MIDI number, so
      // re-centring it on the octave gives a small, musical lean
      // note_lead is 0 until the lead's first note, and a raw 0 would read as
      // "five octaves below A4" and slam the pair sideways on the first frame
      let hasNote = note_lead > 1.0;
      let leanAmt = clamp((note_lead - 69.0) / 24.0, -1.0, 1.0) * 0.22;
      let lean = select(0.0, leanAmt, m == 0 && hasNote);
      // counter-swing on alternate fixtures: a bank all sweeping together
      // reads as one wobbling object, opposed pairs read as a rig
      let dir = select(1.0, -1.0, (i % 2) == 1);
      let ang = -1.5708 + fan + lean + sin(time * rate + fi * 0.9) * 0.34 * dir;
      let tint = hue(fract(baseHue + fi * 0.055 + phase * 0.06));
      let amt = (0.05 + env * 1.35) * drop * live;
      col += tint * beam(p, truss, ang, focus) * amt;
      // the lens itself, so the shaft has a source
      col += tint * flare(p, truss + vec2f(cos(ang), sin(ang)) * 0.10, amt * 0.55);
    }
 
    // KICK: a hard white strobe plus two warm blinders at the truss corners.
    // This one STAYS on hit_, pow'd hard: a strobe is meant to snap, and it is
    // the single element in the rig that should not be smooth. vel_kick scales
    // it so a softer hit gives a softer flash.
    let flash = pow(clamp(hit_kick, 0.0, 1.0), 3.0) * drop * (0.45 + vel_kick * 0.55);
    col += vec3f(1.0, 0.97, 0.93) * flash * 0.55;
    let bl = exp(-length(p - vec2f(-0.82, 0.72)) * 2.6)
           + exp(-length(p - vec2f(0.82, 0.72)) * 2.6);
    col += vec3f(1.0, 0.80, 0.48) * bl * flash * 1.6;
 
    // FLOOR WASH: a pool the beams land in. Cold and shallow in the build,
    // warm and deep after. Leans on the pump rather than on bass alone.
    let floorGlow = smoothstep(0.0, 1.0, (-p.y - 0.30) / 0.70);
    col += mix(vec3f(0.10, 0.16, 0.42), vec3f(0.42, 0.14, 0.62), drop)
         * floorGlow * (0.06 + build * 0.10 + bass * 0.30 + pump * 0.7);
 
    // LED BATTEN along the bottom, fed from the low 60% of the spectrum --
    // above that this patch is mostly air and the bars just sit still. It
    // grows out of the floor through the build, which is the one element that
    // visibly RISES rather than just getting brighter. Stereo-split: the left
    // half of the bar reads the left channel, so the batten shows the image.
    let sideAmt = select(right, left, uv.x < 0.5);
    let s = spectrum(uv.x * 0.6) * (0.25 + build * 0.75);
    let top = -0.94 + s * (0.16 + drop * 0.22) * (0.7 + sideAmt * 0.6);
    let lit = step(p.y, top) * step(-0.985, p.y);
    let cap = smoothstep(0.012, 0.0, abs(p.y - top));
    col += hue(fract(baseHue + uv.x * 0.5 + phase * 0.08)) * (lit * 0.5 + cap)
         * (0.20 + build * 0.25 + level * (0.4 + drop * 0.6));
 
    // LASER: the waveform as one thin line, gated by the lead so it only cuts
    // the room when the lead is sounding. flux adds a flicker on spectral
    // CHANGE -- transitions and edits, not steady notes -- so the line reacts
    // to the patch being altered while it plays.
    let w = waveform(uv.x);
    col += mix(vec3f(0.45, 0.80, 1.0), vec3f(0.35, 1.0, 0.85), drop)
         * smoothstep(0.018, 0.0, abs(p.y - w * 0.42))
         * (0.06 + build * 0.14 + leadE * (0.25 + drop * 0.35) + flux * 0.4) * live;
 
    // POINTER: a soft follow-spot, so the rig can be steered by hand while it
    // runs. click flares it. Idle in the centre, which is where it starts.
    let pd = length(p - (pointer - 0.5) * vec2f(2.0 * asp, 2.0));
    col += vec3f(0.9, 0.85, 1.0) * exp(-pd * 3.2) * (0.05 + click * 0.9) * live;
 
    // room bloom on the beat, then vignette
    let r = length(p);
    col += mix(vec3f(0.20, 0.30, 0.85), vec3f(0.45, 0.18, 0.95), drop)
         * beat * (0.10 + drop * 0.16) / (r * 2.4 + 0.55);
    col *= 1.0 - 0.38 * smoothstep(0.55, 1.7, r);
 
    // Reinhard, so a kick strobe stacked on six beams rolls off instead of
    // clipping to flat white and losing the beams inside it
    col = col / (col + vec3f(1.0));
    return vec4f(pow(col, vec3f(0.85)), 1.0);
  }
`
