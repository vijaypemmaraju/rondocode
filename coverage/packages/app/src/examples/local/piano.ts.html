/* LOCAL ONLY — gitignored, and a production build aliases the loader away.
 *
 * A PIANO, built out of what a piano physically is. Four voices to compare:
 * the same phrase through each, so the thing that matters (does it sound like
 * a piano) can be judged by ear against the thing that is measurable.
 *
 * WHAT MAKES A PIANO A PIANO, in the order it matters:
 *
 * 1. INHARMONICITY. A piano string is STIFF, so its partials are stretched
 *    sharp of the harmonic series:  f_n = n*f0*sqrt(1 + B*n^2).
 *    B runs about 0.00008 in the bass to 0.0008+ in the treble. This is the
 *    single most identifying feature: harmonic partials read as an organ no
 *    matter what envelope you put on them. Measured on partial 8 at B=0.0004:
 *    22.0 cents sharp, against 21.9 predicted.
 * 2. PER-PARTIAL DECAY. High partials die first, so a note gets darker as it
 *    rings — the opposite of a filter sweep, and why a piano note never
 *    sounds static.
 * 3. PITCH-DEPENDENT LENGTH. A bass note rings for 20 seconds, a top-octave
 *    note for under one.
 * 4. VELOCITY IS BRIGHTNESS, not level. A hard strike excites more high
 *    partials; playing louder on a synth that only gets louder reads as a
 *    volume knob, which is the usual tell. Use the `velocity` signal for it,
 *    NOT `gate` — the voice already scales its output by velocity, so `gate`
 *    only re-applies loudness and changes no timbre at all (measured: the
 *    spectral tilt came out identical to four digits).
 * 5. TWO OR THREE STRINGS per note, detuned by a cent or so, which gives the
 *    slow beating and the two-stage decay (fast at first, then a long quiet
 *    aftersound as the strings fall out of phase).
 */
 
const PARTIALS = 16
 
/** Everything above, as one synth. `k` is the partial number. */
const grand = `const grand = synth(({ note, gate, velocity, adsr, sine, noise, svf }) => {
  const f0 = note.freq
  const B = 0.00012
 
  // LENGTH FALLS WITH PITCH. This has to be a SIGNAL, not a number: the graph
  // is built once for the synth, not once per note, so a JS constant here
  // would give a top-octave note the same 8-second ring as the bottom one.
  // 440/f0 is ~8 at a1 and ~0.4 at c6 — roughly the real spread.
  const ring = f0.pow(-0.62).mul(38)
 
  // The strike: a short filtered noise thump, brighter the harder it is hit.
  // Without it the note starts from nothing and reads as a pad.
  const hammer = svf(noise(), velocity.range(1100, 6000), { res: 0.4 })
    .mul(adsr(gate, { a: 0.0005, d: 0.035, s: 0, r: 0.03 }))
    .mul(0.16)
 
  let out = hammer
  for (let k = 1; k <= ${PARTIALS}; k++) {
    const ratio = k * Math.sqrt(1 + B * k * k)
    // partial k lasts roughly 1/k^0.66 of the fundamental's ring
    const d = ring.mul(1 / Math.pow(k, 0.66))
    // ...and the harder the strike, the more of it there is. VELOCITY, not
    // gate: gate is the note's on/off shape, and the voice already scales its
    // own output by velocity, so gate here would only re-apply loudness.
    // This is the line that makes playing harder sound DIFFERENT rather than
    // just bigger, which is where a piano lives or dies.
    const amp = velocity.pow(0.35 + k * 0.09).mul(1 / Math.pow(k, 1.15))
    const env = adsr(gate, { a: 0.002, d, s: 0, r: 0.28 })
    const one = sine(f0.mul(ratio)).mul(env).mul(amp)
    // two strings ~1.2 cents apart on the partials where beating is audible:
    // the slow shimmer, and the two-stage decay as they fall out of phase
    const pair = k <= 6
      ? one.add(sine(f0.mul(ratio * 1.0007)).mul(adsr(gate, { a: 0.002, d: d.mul(1.4), s: 0, r: 0.28 })).mul(amp).mul(0.8))
      : one
    out = out.add(pair)
  }
  // a piano is not a point source: a little air, and the lid
  return out.mul(0.09).tanh()
}, ({ input, reverb }) => input.mix(reverb(input, { roomSize: 0.62, damp: 0.42 }), 0.18))`
 
/** The same thing with the identifying feature REMOVED, to hear what it is
 *  doing. Everything else is identical. */
const organ = grand
  .replace('const grand = synth', 'const flat = synth')
  .replace('const B = 0.00012', 'const B = 0 // harmonic: the tell')
 
/** The cheap comparison: an FM electric piano. Two operators instead of 38
 *  oscillators.
 *
 *  THE FIRST VERSION WAS UNLISTENABLE, and the reason is worth keeping: it ran
 *  a 14:1 modulator at index 7.5 for the WHOLE note. A high ratio at a high
 *  index is a clangorous inharmonic bell, and holding it through the sustain
 *  is a fire alarm. A real DX Rhodes is two jobs: a 1:1 modulator at a modest
 *  index for the BODY, and the high-ratio tine only in the first breath of the
 *  attack, gone before the note settles. */
const rhodes = `const rhodes = synth(({ note, gate, velocity, adsr, fm, sine }) => {
  const f = note.freq
  const env = adsr(gate, { a: 0.002, d: 2.2, s: 0.1, r: 0.45 })
 
  // BODY: 1:1 modulation, index ~1.2 falling to almost nothing. Harmonic, so
  // it thickens the tone instead of ringing against it.
  const bodyIndex = adsr(gate, { a: 0.001, d: 0.9, s: 0.08, r: 0.4 }).mul(velocity.pow(0.6)).mul(1.4)
  const body = fm(f, fm(f).mul(bodyIndex))
 
  // TINE: the metallic ping. High ratio, LOW index, and a 60 ms decay — it is
  // an attack transient, not a timbre. This is the whole difference.
  const tineIndex = adsr(gate, { a: 0.001, d: 0.06, s: 0, r: 0.05 }).mul(velocity).mul(0.9)
  const tine = fm(f.mul(14), fm(f.mul(14)).mul(tineIndex))
    .mul(adsr(gate, { a: 0.0008, d: 0.25, s: 0, r: 0.2 }))
    .mul(velocity.pow(1.4))
    .mul(0.18)
 
  return body.mul(env).add(tine).mul(sine(4.5).mul(0.05).add(1)).mul(0.32)
}, ({ input, reverb }) => input.mix(reverb(input, { roomSize: 0.5, damp: 0.55 }), 0.14))`
 
/** The SAME instrument as `grand`, as one engine node. Everything the additive
 *  version spells out by hand — stretched partials, paired strings, per-partial
 *  decay, ring time falling with pitch — is what the model IS, and it costs a
 *  sixth as much. This is the one rondo can write: `modal model:piano decay:6`. */
const modalPiano = `const modal_piano = synth(({ note, gate, velocity, modal, svf, noise, adsr }) => {
  const strings = modal(gate, note.freq, { model: 'piano', decay: 6 })
  // the hammer, same as grand: velocity is BRIGHTNESS, and the modal bank's
  // own excitation is a bare impulse with no strike noise of its own
  const hammer = svf(noise(), velocity.range(1100, 6000), { res: 0.4 })
    .mul(adsr(gate, { a: 0.0005, d: 0.035, s: 0, r: 0.03 }))
    .mul(0.1)
  // velocity opens the tone as well as the level: a soft strike is duller
  return svf(strings, velocity.range(1400, 11000), { res: 0.1 }).add(hammer).mul(0.7)
}, ({ input, reverb }) => input.mix(reverb(input, { roomSize: 0.62, damp: 0.42 }), 0.18))`
 
const code = `${modalPiano}
 
${grand}
 
${organ}
 
${rhodes}
 
// A phrase with something to judge: spread voicings, a bass that must RING,
// and a dynamic shape, because velocity is where a piano lives or dies.
const LEFT = "<[c2 ~ g2 ~] [f2 ~ c3 ~] [a1 ~ e2 ~] [g1 ~ d2 ~]>"
const RIGHT = "<[e4 g4 c5 g4] [a4 c5 f5 c5] [c5 e5 a5 e5] [b4 d5 g5 d5]>"
const DYN = "0.35 0.55 0.8 0.5 0.95 0.45 0.7 0.4"
 
// swap the .sound() to compare:
//   'modal_piano' — the engine model, one rondo line, ~6x cheaper
//   'grand'       — the additive version it was built from
//   'flat'        — grand with the inharmonicity off (the tell)
//   'rhodes'      — the FM electric, rebuilt
p('lh', note(LEFT).sound('modal_piano').dur(0.9).gain(0.75))
p('rh', note(RIGHT).sound('modal_piano').dur(0.85).gain(DYN))
 
masterGain(9)  // the additive stack lands quiet; this is level, not tone
setCps(0.5)
`
 
export default { name: 'piano (3 ways)', code }
