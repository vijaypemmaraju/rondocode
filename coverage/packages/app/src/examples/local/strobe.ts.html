/* LOCAL ONLY — gitignored, and a production build aliases the loader away.
 *
 * "Strobe" (deadmau5), measures 108-115 of strobe-by-deadmau5.mid.
 *
 * The RONDO source is the truth: it is what is edited in the app, and the
 * rondocode twin is transpiled from it at load, so the two cannot drift. */
 
import { compile } from '@rondocode/rondo'
 
const rondo = `# STROBE (deadmau5) — measures 108-115 of
# strobe-by-deadmau5.mid, transcribed. 84 bpm.
#
# NOTE: the whole 4-bar figure is ONE line with
# \`slow 8\`. Four separate notation lines would
# STACK as simultaneous voices, not play in
# sequence — which is how the first attempt at
# this ended up as four bars at once.
#
# SERUM-ISMS, and why each one is here:
#   detune  — unison WITHOUT detune is just 9 copies of the same pitch. The
#             supersaw character IS the spread; this is the one that matters.
#   unison  — clamps to 9 in the engine, so 9 is the honest number to write.
#   curve 5 — pulls the inner voices toward the centre ("Serum-style focus"),
#             so the stack keeps a defined pitch instead of smearing.
#   w2/w3   — the stacked, slightly-detuned wavetable layers. They were
#             DEFINED and never mixed in, so two thirds of the oscillator
#             section was dead code.
#   drift   — a slow LFO on the wavetable position, on top of the envelope's
#             sweep, so a held note keeps moving instead of freezing.
#   ott     — the multiband glue every Serum patch is printed through.
#   width   — the stereo pull unison spread alone does not give.
 
macro bright 7200 700..7200
 
# DRUMS on/off, from one tap. Both halves have to move together: the duck is
# triggered by kick noteOns, not by the kick's level, so a kick muted to gain 0
# would keep pumping everything else against a beat you cannot hear.
switch drums 1 0
 
synth bell unison:9 detune:.28 spread:1 voices:32 humanize:0.5 curve:5
  wavetable note scan warp:bend table:pwm
  mix w2 0.2
  mix w3 0.2
  inv = 7000 - bright
  svf 534 res:0.3

  dec = bright * 5.6 / 72300
  dec2 = min dec 0.705
    env = adsr .002 dec2 0 0
  base = env -> 0.364..0.63
  scan = base
  * max inv/20000 0
 
  w2 = wavetable note*1 scan table:harmonic warp:mirror warpamt:0.584
  w3 = wavetable note/1 scan table:harmonic

    norm = bright / 8500
    mx = 0.25 - norm * 0.25

# A LIGHT SHOW, not a spectrum analyser. Every element is wired to something
# the arrangement actually does, so it reads as a rig responding to the track
# rather than a meter that happens to wiggle:
#
#   hit_kick  -> the strobe and the blinders. Kick is four-on-the-floor here,
#                so this is the pulse everything else is felt against.
#   hit_lead  -> beams 0 and 3, and the laser scan.
#   hit_bell  -> beams 1 and 4 (the bell plays the same figure as the lead an
#                octave apart, so the two banks fire together and stay a pair).
#   hit_sub   -> beams 2 and 5, the slow ones. Sub moves once per three steps.
#   bass      -> the floor wash. Continuous, so the room never goes fully dark.
#   phase     -> hue rotation. One cycle is one bar, so the colour walks with
#                the 4-bar figure instead of drifting at some unrelated rate.
#
# hit_<name> is an onset envelope with a 0.12 s decay, which is roughly a
# moving head's snap. beat is bass-driven with a 0.18 s decay.
 
visual
  // sRGB-ish hue wheel. Procedural rather than a palette array so a beam
  // index can pick a colour without a dynamically-indexed const array.
  fn hue(h: f32) -> vec3f {
    let k = abs(fract(h + vec3f(0.0, 0.66667, 0.33333)) * 6.0 - 3.0) - 1.0;
    return clamp(k, vec3f(0.0), vec3f(1.0));
  }
 
  // One volumetric beam from \`origin\` at \`ang\`. The cone WIDENS with distance
  // and dims with it, which is the whole reason a beam in haze reads as a
  // beam: a constant-width line looks like a drawn stroke, not a light.
  fn beam(p: vec2f, origin: vec2f, ang: f32) -> f32 {
    let d = vec2f(cos(ang), sin(ang));
    let rel = p - origin;
    let t = dot(rel, d);
    if (t < 0.0) { return 0.0; }
    let perp = length(rel - d * t);
    let w = 0.035 + t * 0.075;
    return exp(-(perp * perp) / (w * w)) * exp(-t * 0.5);
  }
 
  fn render(uv: vec2f) -> vec4f {
    let asp = res.x / max(res.y, 1.0);
    let p = (uv - 0.5) * vec2f(2.0 * asp, 2.0);
    var col = vec3f(0.0);
 
    // haze: denser low in the room, and it swells on the bass so the air
    // itself pumps rather than only the fixtures
    let low = smoothstep(1.0, -1.0, p.y);
    col += vec3f(0.045, 0.06, 0.11) * (0.35 + low) * (1.0 + bass * 1.8);
 
    // the truss sits above frame, so the beams enter already fanned
    let truss = vec2f(0.0, 1.3);
    for (var i: i32 = 0; i < 6; i++) {
      let fi = f32(i);
      // each bank gets its own envelope; 0/3 lead, 1/4 bell, 2/5 sub
      let m = i % 3;
      var env = hit_lead;
      if (m == 1) { env = hit_bell; }
      if (m == 2) { env = hit_sub; }
      // fan, then sweep. The sub pair sweeps at a third the rate, so the rig
      // has slow movement under the fast movement instead of one busy speed.
      let rate = select(0.55, 0.18, m == 2);
      let fan = (fi - 2.5) * 0.245;
      let ang = -1.5708 + fan + sin(time * rate + fi * 1.15) * 0.30;
      // idle glow so the rig is visible between notes, then flare on the hit
      col += hue(fract(0.56 + fi * 0.075 + phase * 0.12))
           * beam(p, truss, ang) * (0.10 + env * 1.7);
    }
 
    // KICK: a hard white strobe plus two warm blinders at the truss corners.
    // Pow'd hard so it is a snap, not a throb -- a linear envelope here reads
    // as the whole screen breathing, which is the opposite of a strobe.
    let flash = pow(clamp(hit_kick, 0.0, 1.0), 2.5);
    col += vec3f(1.0, 0.97, 0.93) * flash * 0.42;
    let bl = exp(-length(p - vec2f(-0.82, 0.72)) * 2.6)
           + exp(-length(p - vec2f(0.82, 0.72)) * 2.6);
    col += vec3f(1.0, 0.78, 0.45) * bl * flash * 1.1;
 
    // floor wash: a bass-driven pool the beams appear to land in
    let floorGlow = smoothstep(0.0, 1.0, (-p.y - 0.30) / 0.70);
    col += vec3f(0.25, 0.10, 0.55) * floorGlow * (0.10 + bass * 0.9);
 
    // LED batten along the bottom, fed from the low 60% of the spectrum --
    // above that this patch is mostly air and the bars just sit still
    let s = spectrum(uv.x * 0.6);
    let top = -0.94 + s * 0.30;
    let lit = step(p.y, top) * step(-0.985, p.y);
    let cap = smoothstep(0.012, 0.0, abs(p.y - top));
    col += hue(fract(uv.x * 0.7 + phase * 0.18)) * (lit * 0.5 + cap) * (0.35 + level);
 
    // laser scan: the waveform as a single thin line, gated by the lead so it
    // only cuts the room when the lead is actually playing
    let w = waveform(uv.x);
    col += vec3f(0.35, 1.0, 0.85)
         * smoothstep(0.018, 0.0, abs(p.y - w * 0.42)) * (0.10 + hit_lead * 0.55);
 
    // room bloom on the beat, then vignette
    let r = length(p);
    col += vec3f(0.42, 0.16, 0.95) * beat * 0.22 / (r * 2.4 + 0.55);
    col *= 1.0 - 0.38 * smoothstep(0.55, 1.7, r);
 
    // Reinhard, so a kick strobe stacked on six beams rolls off instead of
    // clipping to flat white and losing the beams inside it
    col = col / (col + vec3f(1.0));
    return vec4f(pow(col, vec3f(0.85)), 1.0);
  }
`
