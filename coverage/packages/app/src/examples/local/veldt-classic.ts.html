import type { Example } from '../index'
 
/* LOCAL example - gitignored, never shipped. THE VELDT, recovered from git
 * history (snapshot 3c0b5c2) as the classic supersaw-palette build; the
 * wavetable rebuild lives in veldt.ts once matched by measurement. */
const code = `// THE VELDT, FULL TRACK. The whole palette: transcribed chord-stab sequence
// + high arp hook + real chord comp + fill lick (ALL FOUR MIDI tracks
// transcribed), on lush synths; sidechain, per-synth compression for
// punch/glue, arranged intro -> build -> drop. A major, 128 BPM. No samples.
 
// ---- drums ----
const kick = synth(({ gate, adsr, sine, noise, svf, compress }) => {
  const pitch = adsr(gate, { a: 0.001, d: 0.09, s: 0, r: 0.05 })
  const amp = adsr(gate, { a: 0.001, d: 0.17, s: 0, r: 0.06 })
  const body = sine(pitch.pow(3).range(46, 190)).mul(amp)
  const click = svf(noise(), 4200, { mode: 'hp' }).mul(adsr(gate, { a: 0.0004, d: 0.012, s: 0, r: 0.004 })).mul(0.4)
  // compress the kick for a tight, punchy transient
  return compress(body.add(click).tanh(), { threshold: -16, ratio: 4, attack: 3, release: 90, makeup: 2 })
})
const clap = synth(({ gate, adsr, noise, svf }) =>
  svf(noise(), 1700, { mode: 'bp', res: 0.5 }).mul(adsr(gate, { a: 0.003, d: 0.16, s: 0, r: 0.09 })).mul(1.2).tanh())
const hat = synth(({ gate, adsr, noise, svf }) =>
  svf(noise(), 8800, { mode: 'hp' }).mul(adsr(gate, { a: 0.001, d: 0.05, s: 0, r: 0.03 })).mul(0.8))
 
// ---- deep sub (plays the stab's bass voice an octave below) ----
const sub = synth(({ note, gate, adsr, sine, saw, onepole }) => {
  const env = adsr(gate, { a: 0.006, d: 0.12, s: 0.85, r: 0.08 })
  const f = note.freq.mul(0.5) // octave down
  const tone = sine(f).mix(saw(f), 0.22)
  return onepole(tone, 260).mul(env).mul(0.6).tanh().mul(0.85)
}, ({ input }) => input, { mono: true, glide: 0.03, voices: 1 })
 
// ---- THE chord stab: 5-voice supersaw, env+LFO filter, glued with compress ----
const stab = synth(({ note, gate, param, adsr, saw, svf, lfo }) => {
  const bright = param('bright', 8800, { min: 2500, max: 13000, curve: 'log' })
  const env = adsr(gate, { a: 0.002, d: 0.16, s: 0.15, r: 0.09 })
  const f = note.freq
  const osc = saw(f).mix(saw(f.mul(1.004)), 0.5).mix(saw(f.mul(2)), 0.34)
  const cut = env.range(0.5, 1).mul(lfo(0.07).range(bright.mul(0.75), bright))
  return svf(osc, cut, { res: 0.42 }).mul(env).mul(0.22)
}, ({ input, chorus, reverb, compress }) => {
  const w = chorus(input, { rate: 0.5, depth: 0.003, mix: 0.35 })
  const wet = w.mix(reverb(w, { roomSize: 0.86, damp: 0.3 }), 0.32)
  // bus glue: hold the stack together
  return compress(wet, { threshold: -20, ratio: 3, attack: 8, release: 120, makeup: 3 })
}, { unison: 5, detune: 12, spread: 0.8, voices: 18 })
 
// ---- the high arp HOOK (the melodic top line, from the MIDI) ----
const lead = synth(({ note, gate, adsr, wavetable, svf }) => {
  const env = adsr(gate, { a: 0.003, d: 0.157, s: 0.256, r: 0.071 })
  const osc = wavetable(note.freq, 0.334, { table: 'harmonic' })
  return svf(osc, env.range(0.294, 0.5).mul(3600), { res: 0.238 }).mul(env).mul(0.735)
}, ({ input, chorus, delay, reverb }) => {
  const w = chorus(input, { rate: 0.565, depth: 0.003, mix: 0.35 })
  const ech = delay(w, 0.234, 0.3) // 8th-note echo at 128 BPM
  const wet = w.add(ech.mul(0.35))
  return wet.mix(reverb(wet, { roomSize: 0.85, damp: 0.3 }), 0.13)
}, { unison: 2, detune: 10, spread: 0.61, voices: 7 })
 
// ---- lush supersaw PAD: the intro chords (the emotional Veldt opening) ----
const pad = synth(({ note, gate, adsr, saw, svf, lfo }) => {
  const env = adsr(gate, { a: 0.4, d: 0.7, s: 0.9, r: 1.5 })
  const f = note.freq
  const osc = saw(f).mix(saw(f.mul(1.004)), 0.5).mix(saw(f.mul(2.002)), 0.16)
  return svf(osc, lfo(0.05).range(800, 2200), { res: 0.14 }).mul(env).mul(0.14)
}, ({ input, chorus, reverb }) => {
  const w = chorus(input, { rate: 0.35, depth: 0.005, mix: 0.5 })
  return w.mix(reverb(w, { roomSize: 0.92, damp: 0.4 }), 0.42)
}, { unison: 7, detune: 13, spread: 0.9, voices: 24 })
// ---- synth riser (noise + rising highpass) for the build transition ----
const riser = synth(({ gate, adsr, noise, svf }) => {
  const env = adsr(gate, { a: 3.2, d: 0, s: 1, r: 0.2 })
  return svf(noise(), env.range(400, 7000), { mode: 'hp', res: 0.4 }).mul(env.pow(2)).mul(0.28)
})
// ---- bright pluck for the MIDI "Bassline" fill lick (octave-doubled) ----
const fill = synth(({ note, gate, adsr, saw, svf }) => {
  const env = adsr(gate, { a: 0.002, d: 0.14, s: 0, r: 0.1 })
  const osc = saw(note.freq).mix(saw(note.freq.mul(2)), 0.3)
  return svf(osc, env.range(0.4, 1).mul(7200), { res: 0.3 }).mul(env).mul(0.4)
}, ({ input, delay, reverb }) => {
  const w = input.add(delay(input, 0.234, 0.3).mul(0.35)) // 8th-note echo
  return w.mix(reverb(w, { roomSize: 0.85, damp: 0.3 }), 0.34)
})
 
// ==== harmony ====
// the chord-stab sequence (three voices, 8th notes, 8-bar loop), the drop body
const s0 = note('<[e2 e2 e2 d2 d2 d2 d2 d2] [f#2 f#2 f#2 a2 a2 a2 a2 a2] [b2 b2 b2 a2 a2 a2 d2 d2] [d2 d2 d2 d2 d2 d2 d2 d2] [e2 e2 e2 b1 b1 b1 b1 b1] [f#2 f#2 f#2 e2 e2 a2 a2 a2] [b2 b2 b2 a2 a2 a2 d2 d2] [d2 d2 d2 d2 d2 d2 d2 d2]>')
const s1 = note('<[g#3 g#3 g#3 a3 a3 a3 a3 g#3] [c#4 b3 c#4 b3 b3 a3 a3 a3] [d4 d4 d4 c#4 c#4 c#4 a3 a3] [a3 a3 a3 a3 a3 a3 a3 a3] [g#3 g#3 g#3 d3 d3 d3 d3 g#3] [a3 b3 a3 g#3 g#3 c#4 c#4 c#4] [d4 d4 d4 c#4 c#4 c#4 f#3 f#3] [f#3 f#3 f#3 f#3 f#3 f#3 f#3 f#3]>')
const s2 = note('<[e4 e4 e4 e4 e4 e4 e4 e4] [e4 e4 e4 e4 e4 e4 e4 e4] [f#4 f#4 f#4 e4 e4 e4 e4 e4] [e4 e4 e4 e4 e4 e4 e4 e4] [e4 e4 e4 e4 e4 e4 e4 e4] [e4 e4 e4 e4 e4 e4 e4 e4] [f#4 f#4 f#4 e4 e4 e4 a3 a3] [a3 a3 a3 a3 a3 a3 a3 a3]>')
const stabSeq = stack(s0, s1, s2)
// the REAL chord comp, transcribed from the MIDI "Piano" track (track 5):
// the actual E-D-F#m-A-B voicing + rhythm (the ~ rests make it breathe).
const q0 = note('<[e2 ~ ~ d2 ~ ~ ~ ~] [f#2 b3 c#4 a2 ~ ~ a3 ~] [b2 ~ ~ a2 ~ ~ d2 ~] [~ ~ ~ ~ ~ ~ ~ ~] [e2 ~ ~ b1 ~ ~ ~ g#3] [f#2 b3 a3 e2 ~ a2 ~ ~] [b2 ~ ~ a2 ~ ~ d2 ~] [~ ~ ~ ~ ~ ~ ~ ~]>')
const q1 = note('<[g#3 ~ ~ a3 ~ ~ ~ ~] [c#4 b3 c#4 b3 ~ ~ e4 ~] [d4 ~ ~ c#4 ~ ~ a3 ~] [~ ~ ~ ~ ~ ~ ~ ~] [g#3 ~ ~ d3 ~ ~ ~ e4] [a3 b3 a3 g#3 ~ c#4 ~ ~] [d4 ~ ~ c#4 ~ ~ f#3 ~] [~ ~ ~ ~ ~ ~ ~ ~]>')
const q2 = note('<[e4 ~ ~ e4 ~ ~ ~ ~] [e4 b3 c#4 e4 ~ ~ e4 ~] [f#4 ~ ~ e4 ~ ~ e4 ~] [~ ~ ~ ~ ~ ~ ~ ~] [e4 ~ ~ e4 ~ ~ ~ e4] [e4 b3 a3 e4 ~ e4 ~ ~] [f#4 ~ ~ e4 ~ ~ a3 ~] [~ ~ ~ ~ ~ ~ ~ ~]>')
const piano = stack(q0, q1, q2)
// the high arp hook
const hook = note('<[a5 b5 e6 a5 b5 f#6 a5 b5] [e6 a5 b5 f#6 a5 b5 e6 a5] [b5 f#6 a5 b5 e6 a5 b5 a6] [a5 b5 g#6 a5 b5 e6 a5 b6]>')
// the MIDI "Bassline" track, a one-bar fill lick (bar 7 of the 8-bar loop),
// octave-doubled: D D D C# - A B -
const fillLo = note('<~ ~ ~ ~ ~ ~ [d5 d5 d5 c#5 ~ a4 b4 ~] ~>')
const fillHi = note('<~ ~ ~ ~ ~ ~ [d6 d6 d6 c#6 ~ a5 b5 ~] ~>')
const fillLick = stack(fillLo, fillHi)
 
// ==== reusable layers ====
const kickFour = note('c1*4').sound('kick')
const hatsOff = note('~ c5 ~ c5 ~ c5 ~ c5').sound('hat')
const clapBack = note('~ c2 ~ c2').sound('clap')
const subBass = stack(s0).sound('sub').dur(0.12)
 
// ==== SECTIONS, the full song arc ====
// Structure (radio-edit style): the STAB SEQUENCE opens the track and cooks;
// the drums come in over it; only THEN does the arp/hook lead arrive.
// 1. intro: the chord-stab sequence alone (+ pad bed), no beat, no lead
const intro = stack(
  stabSeq.sound('stab').dur(0.115).gain(0.72),
  piano.sound('pad').dur(0.9).gain(0.7),
)
// 2. build: the drums come in over the cooking stabs; sub + riser. Still no lead.
const build = stack(
  stabSeq.sound('stab').dur(0.115).gain(0.92),
  piano.sound('pad').dur(0.9).gain(0.55),
  kickFour.gain(0.9),
  hatsOff.gain(rand.range(0.4, 0.7)),
  subBass.gain(1.0),
  note('c4').slow(4).sound('riser'),
)
// 3. drop: the LEAD/hook finally arrives on top of the driving stabs + full beat
const drop = stack(
  stabSeq.sound('stab').dur(0.115).gain(1.0),
  piano.sound('pad').dur(0.9).gain(0.5),
  hook.sound('lead').gain(1.25),
  kickFour.gain(1.05),
  clapBack,
  hatsOff.gain(rand.range(0.5, 0.85)),
  subBass.gain(1.3),
  fillLick.sound('fill').dur(0.12).gain(0.75),
)
// 4. breakdown: strip to pad + hook, drenched, the emotional breather
const breakdown = stack(
  piano.sound('pad').dur(0.95).gain(1.0),
  hook.sound('lead').gain(0.6),
)
 
// arrange: stabs open -> drums join (stabs cook) -> DROP (lead arrives) ->
// breakdown -> build -> DROP2 -> outro  (52 bars, ~1:40 at 128 BPM)
p('veldt', arrange(
  [8, intro],       // stabs open the track and cook alone
  [8, build],       // drums come in over the stabs, still no lead
  [8, drop],        // the arp/hook lead finally arrives: the payoff
  [8, breakdown],   // strip back to pad + hook
  [8, build],       // rebuild
  [8, drop],        // second drop
  [4, breakdown],   // outro tail
))
// sidechain pump: kick ducks the melodic + low elements (not the drums)
sidechain('kick', { depth: 0.7, release: 0.16, duck: { stab: 0.5, pad: 0.5, sub: 0.9, lead: 0.4 } })
// master glue: hold the whole mix together across the sections
masterCompress({ threshold: -16, ratio: 3, attack: 12, release: 140, makeup: 2 })
setCps(0.533)
 
// a calm dusk-over-the-veldt scene: an indigo sky the pad warms, a low sun the
// chord stab makes breathe, a shimmering spectrum horizon, and soft golden
// sparks from the lead arp. Smooth motion, gentle swells, no strobing.
visual(\`
fn render(uv: vec2f) -> vec4f {
  let p = (uv * 2.0 - 1.0) * vec2f(res.x / res.y, 1.0);
  // deep dusk sky, indigo up top easing to a warm band down low
  let sky = mix(vec3f(0.05, 0.06, 0.14), vec3f(0.17, 0.10, 0.20), uv.y);
  // the PAD warms the whole sky, gently
  let warm = vec3f(0.20, 0.10, 0.16) * (0.4 + hit_pad * 0.4);
  // a soft sun low on the horizon; the STAB makes it breathe a little
  let sun = p - vec2f(0.0, -0.35);
  let glow = (0.10 + hit_stab * 0.12) / (dot(sun, sun) * 2.2 + 0.18);
  // a shimmering spectrum ridge along the horizon line
  let ridge = 0.32 + spectrum(fract(uv.x * 1.5)) * 0.28;
  let horizon = smoothstep(0.06, 0.0, abs(p.y + 0.35 - ridge)) * (0.5 + level * 0.4);
  // the LEAD arp scatters soft golden sparks across the upper sky
  let spk = pow(0.5 + 0.5 * sin(uv.x * 40.0 + time * 1.5), 6.0)
          * smoothstep(0.0, 0.6, uv.y) * hit_lead * 0.5;
  // a gentle waveform aurora ribbon
  let ribbon = smoothstep(0.03, 0.0, abs(p.y - 0.25 - waveform(uv.x) * 0.18)) * 0.5;
  let col = sky + warm
          + vec3f(1.0, 0.7, 0.4) * glow
          + vec3f(0.9, 0.6, 0.7) * horizon
          + vec3f(1.0, 0.85, 0.5) * spk
          + vec3f(0.4, 0.8, 0.9) * ribbon * (0.4 + hit_lead * 0.3);
  return vec4f(min(col, vec3f(1.0)), 1.0);
}
\`)
`
